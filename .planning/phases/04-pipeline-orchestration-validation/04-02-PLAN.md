```python
#!/usr/bin/env python3
"""Generate reference OCR output from PaddleX for validation against iOS pipeline.

Usage:
    python3 generate_reference.py [--images-dir DIR] [--output-dir DIR]

Requires: paddleocr >= 3.4.0, paddlex >= 3.4.0
"""

import argparse
import json
import os
import sys
from pathlib import Path


def generate_reference(images_dir: str, output_dir: str) -> None:
    """Run PaddleX OCR on each image and save reference JSON."""
    from paddleocr import PaddleOCR

    # Initialize with PP-OCRv5 mobile models
    ocr = PaddleOCR(
        text_detection_model_name="PP-OCRv5_mobile_det",
        text_recognition_model_name="PP-OCRv5_mobile_rec",
        use_doc_orientation_classify=False,
        use_doc_unwarping=False,
        use_textline_orientation=False,
    )

    images_path = Path(images_dir)
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    image_extensions = {".jpg", ".jpeg", ".png", ".bmp", ".tiff"}
    image_files = sorted(
        f for f in images_path.iterdir()
        if f.suffix.lower() in image_extensions
    )

    if not image_files:
        print(f"Error: No images found in {images_dir}", file=sys.stderr)
        sys.exit(1)

    print(f"Processing {len(image_files)} images...")

    for img_file in image_files:
        print(f"  {img_file.name}...", end=" ", flush=True)
        results = list(ocr.predict(str(img_file)))

        # PaddleOCR.predict() returns a list of result objects.
        # Each result has .boxes (list of dict with 'rec_text', 'rec_score', 'dt_polys')
        # or uses the attribute-based access pattern.
        boxes = []
        for result in results:
            # Access detection polygons and recognition results
            if hasattr(result, "rec_texts"):
                # Newer PaddleOCR API
                for i in range(len(result.rec_texts)):
                    poly = result.dt_polys[i].tolist() if hasattr(result.dt_polys[i], "tolist") else result.dt_polys[i]
                    boxes.append({
                        "polygon": [[int(p[0]), int(p[1])] for p in poly],
                        "text": result.rec_texts[i],
                        "confidence": float(result.rec_scores[i]),
                    })
            elif hasattr(result, "__getitem__"):
                # Legacy API: list of [boxes, (texts, scores)]
                for item in result:
                    if isinstance(item, dict):
                        poly = item.get("dt_polys", item.get("points", []))
                        if hasattr(poly, "tolist"):
                            poly = poly.tolist()
                        boxes.append({
                            "polygon": [[int(p[0]), int(p[1])] for p in poly],
                            "text": item.get("rec_text", ""),
                            "confidence": float(item.get("rec_score", 0.0)),
                        })

        reference = {
            "image": img_file.name,
            "box_count": len(boxes),
            "boxes": boxes,
        }

        out_file = output_path / f"{img_file.stem}.json"
        with open(out_file, "w", encoding="utf-8") as f:
            json.dump(reference, f, indent=2, ensure_ascii=False)

        print(f"{len(boxes)} boxes")

    print(f"\nReference files written to {output_dir}")


def main():
    parser = argparse.ArgumentParser(
        description="Generate PaddleX OCR reference output for iOS validation"
    )
    parser.add_argument(
        "--images-dir",
        default="deploy/ios_demo/Validation/test_images",
        help="Directory containing test images (default: deploy/ios_demo/Validation/test_images)",
    )
    parser.add_argument(
        "--output-dir",
        default="deploy/ios_demo/Validation/reference",
        help="Directory for reference JSON output (default: deploy/ios_demo/Validation/reference)",
    )
    args = parser.parse_args()
    generate_reference(args.images_dir, args.output_dir)


if __name__ == "__main__":
    main()
```

  <action>
Create `validate.py` that compares iOS-exported JSON against reference JSON.

The comparison rules (from RESEARCH.md Pitfall 6):
- **Polygon coordinates**: Exact integer match. If any coordinate differs by even 1, report as FAIL with diff.
- **Recognized text**: Exact string match. Character-by-character comparison.
- **Confidence scores**: Tolerance of 1e-4 (0.0001). Float differences within this range are acceptable due to ARM vs x86 floating-point differences.
- **Box count**: Must match exactly between iOS and reference.
- **Box order**: Boxes are compared in order (both should be in reading order from BoxSorter).

```python
#!/usr/bin/env python3
"""Validate iOS OCR pipeline output against PaddleX reference output.

Usage:
    python3 validate.py [--ios-dir DIR] [--reference-dir DIR] [--verbose]

Exit codes:
    0 = all images match
    1 = at least one mismatch found
    2 = missing files or other error
"""

import argparse
import json
import sys
from pathlib import Path

CONFIDENCE_TOLERANCE = 1e-4


def compare_polygons(ios_poly, ref_poly):
    """Compare two polygons (list of [x, y] integer pairs). Returns list of diffs."""
    diffs = []
    if len(ios_poly) != len(ref_poly):
        diffs.append(f"Point count: iOS={len(ios_poly)} ref={len(ref_poly)}")
        return diffs
    for i, (ios_pt, ref_pt) in enumerate(zip(ios_poly, ref_poly)):
        if ios_pt[0] != ref_pt[0] or ios_pt[1] != ref_pt[1]:
            diffs.append(f"Point {i}: iOS={ios_pt} ref={ref_pt}")
    return diffs


def compare_boxes(ios_box, ref_box, index):
    """Compare a single box pair. Returns list of issue strings."""
    issues = []

    # Polygon comparison (exact integer match)
    poly_diffs = compare_polygons(ios_box["polygon"], ref_box["polygon"])
    if poly_diffs:
        issues.append(f"  Box {index} polygon: {'; '.join(poly_diffs)}")

    # Text comparison (exact string match)
    if ios_box["text"] != ref_box["text"]:
        issues.append(
            f"  Box {index} text: iOS='{ios_box['text']}' ref='{ref_box['text']}'"
        )

    # Confidence comparison (within tolerance)
    ios_conf = ios_box["confidence"]
    ref_conf = ref_box["confidence"]
    if abs(ios_conf - ref_conf) > CONFIDENCE_TOLERANCE:
        issues.append(
            f"  Box {index} confidence: iOS={ios_conf:.6f} ref={ref_conf:.6f} "
            f"(diff={abs(ios_conf - ref_conf):.6f}, tolerance={CONFIDENCE_TOLERANCE})"
        )

    return issues


def validate(ios_dir: str, reference_dir: str, verbose: bool = False) -> bool:
    """Compare iOS output against reference. Returns True if all match."""
    ios_path = Path(ios_dir)
    ref_path = Path(reference_dir)

    if not ref_path.exists():
        print(f"Error: Reference directory not found: {reference_dir}", file=sys.stderr)
        print("Run generate_reference.py first.", file=sys.stderr)
        sys.exit(2)

    if not ios_path.exists():
        print(f"Error: iOS output directory not found: {ios_dir}", file=sys.stderr)
        print("Export iOS pipeline results as JSON first.", file=sys.stderr)
        sys.exit(2)

    ref_files = sorted(ref_path.glob("*.json"))
    if not ref_files:
        print(f"Error: No reference JSON files in {reference_dir}", file=sys.stderr)
        sys.exit(2)

    all_pass = True
    total_images = 0
    total_boxes = 0
    matched_boxes = 0

    for ref_file in ref_files:
        ios_file = ios_path / ref_file.name
        if not ios_file.exists():
            print(f"MISSING: {ref_file.name} -- no iOS output found")
            all_pass = False
            continue

        with open(ref_file, "r", encoding="utf-8") as f:
            ref_data = json.load(f)
        with open(ios_file, "r", encoding="utf-8") as f:
            ios_data = json.load(f)

        total_images += 1
        image_name = ref_data["image"]
        ref_boxes = ref_data["boxes"]
        ios_boxes = ios_data["boxes"]

        if len(ios_boxes) != len(ref_boxes):
            print(
                f"FAIL: {image_name} -- box count: iOS={len(ios_boxes)} ref={len(ref_boxes)}"
            )
            all_pass = False
            continue

        image_issues = []
        for i, (ios_box, ref_box) in enumerate(zip(ios_boxes, ref_boxes)):
            total_boxes += 1
            box_issues = compare_boxes(ios_box, ref_box, i)
            if box_issues:
                image_issues.extend(box_issues)
            else:
                matched_boxes += 1

        if image_issues:
            print(f"FAIL: {image_name}")
            for issue in image_issues:
                print(issue)
            all_pass = False
        else:
            if verbose:
                print(f"PASS: {image_name} ({len(ref_boxes)} boxes)")

    # Summary
    print(f"\n{'='*50}")
    print(f"Images: {total_images}")
    print(f"Boxes:  {matched_boxes}/{total_boxes} matched")
    print(f"Result: {'ALL PASS' if all_pass else 'FAIL'}")
    print(f"{'='*50}")

    return all_pass


def main():
    parser = argparse.ArgumentParser(
        description="Validate iOS OCR output against PaddleX reference"
    )
    parser.add_argument(
        "--ios-dir",
        default="deploy/ios_demo/Validation/ios_output",
        help="Directory containing iOS output JSON (default: deploy/ios_demo/Validation/ios_output)",
    )
    parser.add_argument(
        "--reference-dir",
        default="deploy/ios_demo/Validation/reference",
        help="Directory containing reference JSON (default: deploy/ios_demo/Validation/reference)",
    )
    parser.add_argument(
        "--verbose", "-v",
        action="store_true",
        help="Print PASS results too",
    )
    args = parser.parse_args()

    success = validate(args.ios_dir, args.reference_dir, args.verbose)
    sys.exit(0 if success else 1)


if __name__ == "__main__":
    main()
```

```markdown
# Validation: iOS Pipeline vs Python Reference

Validates that the iOS OCR pipeline produces numerically exact results matching the PaddleX Python reference.

## Prerequisites

- Python 3.8+
- `paddleocr >= 3.4.0` (pip install paddleocr)
- Test images in `test_images/` directory

## Usage

### Step 1: Add test images

Place test images (JPG, PNG) in `test_images/`.

### Step 2: Generate reference output

python3 generate_reference.py

Outputs reference JSON files to `reference/`.

### Step 3: Run iOS pipeline

Run the iOS app on the same test images and export results as JSON to `ios_output/`.
Each file should match the reference filename (e.g., `test_001.json`).

### Step 4: Validate

python3 validate.py
python3 validate.py --verbose  # show PASS results too

## JSON Schema

Each JSON file (both reference and iOS output) follows this schema:

{
  "image": "filename.jpg",
  "box_count": 5,
  "boxes": [
    {
      "polygon": [[x1,y1], [x2,y2], [x3,y3], [x4,y4]],
      "text": "recognized text",
      "confidence": 0.9876
    }
  ]
}

## Match Criteria

- **Polygons**: Exact integer coordinate match
- **Text**: Exact string match
- **Confidence**: Within tolerance of 0.0001 (ARM vs x86 float differences)
- **Box count**: Must match exactly
- **Box order**: Compared in array order (both in reading order)
```
